// ========================================== // VSCMG 核心动力学仿真引擎 (基于欧拉法数值积分) // 契合《平衡模块技术方案》中的倒立摆防抖模型 // ========================================== const DynamicsEngine = { dt: 0.02, // 仿真步长 50Hz (0.02秒) phi: 0, // 当前车身倾角 (Roll angle, 弧度) omega: 0, // 当前角速度 (Roll rate, 弧度/秒) integral_e: 0, // PID 积分误差 // 物理参数 (可根据你的 SolidWorks 实际数据微调) g: 9.81, h: 0.35, // 重心高度 (m) I_sys: 8.5, // 系统的等效转动惯量 (kg·m²) // VSCMG 闭环 PID 控制参数 (模拟论文中的抗扰动控制律) Kp: 1800, // 比例增益：决定拉回力道的刚性 Kd: 250, // 微分增益：提供阻尼，防止震荡 Ki: 50, // 积分增益：消除稳态误差 max_torque: 200, // VSCMG 陀螺仪最大输出补偿力矩 (N·m) // 步进解算函数 step: function(mass, external_dist_torque) { // 1. 倒立摆重力失稳力矩：T_g = m * g * h * sin(phi) let T_g = mass * this.g * this.h * Math.sin(this.phi); // 2. 传感器采样与 PID 控制器计算 let error = 0 - this.phi; // 目标倾角为 0 this.integral_e += error * this.dt; // 积分抗饱和限幅 (Anti-windup) this.integral_e = Math.max(-0.5, Math.min(0.5, this.integral_e)); let T_cmd = this.Kp * error - this.Kd * this.omega + this.Ki * this.integral_e; // VSCMG 物理输出限幅 (电机和飞轮的极限) T_cmd = Math.max(-this.max_torque, Math.min(this.max_torque, T_cmd)); // 3. 系统总力矩 = 重力矩 + 外部瞬态冲击矩 + VSCMG补偿矩 let net_torque = T_g + external_dist_torque + T_cmd; // 4. 动力学解算 (欧拉法积分) let alpha = net_torque / this.I_sys; // 角加速度 = 力矩 / 转动惯量 this.omega += alpha * this.dt; // 更新角速度 this.phi += this.omega * this.dt; // 更新角度 // 引入微小的 MEMS 传感器白噪声，让显示更符合真实硬件 let noise = (Math.random() - 0.5) * 0.001; return { roll: this.phi + noise, roll_rate: this.omega + noise, torque_out: -T_cmd // 显示系统为了抵抗干扰付出的反向力矩 }; } }; // ========================================== // UI 交互与图表渲染逻辑 // ========================================== const viewer = document.getElementById('viewer'); const infoCard = document.getElementById('infoCard'); let simTimer = null; let currentShock = 0; // 当前施加的瞬态冲击力 // 存储历史数据，用于绘制真实的衰减波形 (存储50个点) const historyData = new Array(50).fill(0); // 切换视图逻辑 (保持不变) function switchView(mode) { const isExplode = mode === 'explode'; viewer.src = isExplode ? "model_exploded.glb" : "model.glb"; if(isExplode) { viewer.removeAttribute('auto-rotate'); document.getElementById('btnExplode').classList.add('active'); document.getElementById('btnModel').classList.remove('active'); } else { viewer.setAttribute('auto-rotate', ''); document.getElementById('btnModel').classList.add('active'); document.getElementById('btnExplode').classList.remove('active'); infoCard.style.display = 'none'; } } // 仅仅更新UI数字，不立刻改变物理状态 function updateSimulation() { const dist = parseInt(document.getElementById('distRange').value); document.getElementById('distVal').innerText = `${dist} N·m`; } // 核心：启动 50Hz 的高频仿真循环 function startSimulationLoop() { if(simTimer) clearInterval(simTimer); simTimer = setInterval(() => { const mass = parseFloat(document.getElementById('massInput').value) || 0; // 获取并解算当前的一步物理状态 let state = DynamicsEngine.step(mass, currentShock); // 将弧度转换为角度显示 let rollDeg = state.roll * 180 / Math.PI; let rateDeg = state.roll_rate * 180 / Math.PI; // 更新控制台数据 document.getElementById('torqueOut').innerHTML = `${Math.abs(state.torque_out).toFixed(2)} N·m`; document.getElementById('velOut').innerHTML = `${Math.abs(state.roll_rate).toFixed(3)} rad/s`; document.getElementById('gyroX').innerText = `${rollDeg.toFixed(2)}°`; document.getElementById('gyroY').innerText = `${(rateDeg * 0.05).toFixed(2)}°`; // 模拟Y轴耦合微动 document.getElementById('gyroZ').innerText = `0.00°`; // 绘制真实的稳定性收敛曲线 historyData.push(state.roll); historyData.shift(); drawCurve(); // 模拟外部冲击力矩的自然衰减 (冲击发生后迅速消失) currentShock *= 0.85; }, 20); // 20ms = 50Hz } // 将物理姿态实时映射到 SVG 曲线 function drawCurve() { const path = document.getElementById('stableLine'); let d = "M0 40 "; for(let i = 0; i < historyData.length; i++) { let x = i * (200 / 49); // 将弧度微小变化放大到 SVG Y 轴 (假设偏转幅度在 ±0.2 弧度左右) let y = 40 - (historyData[i] * 300); // 限幅防止曲线画出框外 y = Math.max(5, Math.min(75, y)); d += `L ${x.toFixed(1)} ${y.toFixed(1)} `; } path.setAttribute('d', d); } // 点击按钮，注入外部干扰 function triggerPulse() { const dist = parseInt(document.getElementById('distRange').value); currentShock = dist; // 瞬间注入干扰力矩 // UI反馈 const path = document.getElementById('stableLine'); path.style.stroke = "#f59e0b"; // 曲线变橙色警告 setTimeout(() => { path.style.stroke = "#3b82f6"; }, 500); // 恢复蓝色 } window.onload = () => { updateSimulation(); startSimulationLoop(); // 页面加载即启动物理引擎 };
